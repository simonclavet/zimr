//! Why does a STATIC quadruped slide? And would friction fix it?
//!
//! ★★★ THE DECIDING QUESTION IS THE FRICTION DEMAND WHILE NOTHING MOVES. If the feet creep while
//! the tangential force is well INSIDE the cone, the sliding is not Coulomb slip and more
//! friction cannot fix it — it would be solver creep, or four legs quietly fighting each other.
//! If the demand is AT the cone, friction would help but would be hiding the real cause.
const std = @import("std");
const Allocator = std.mem.Allocator;
const zm = @import("zm");
const rbt = @import("robot.zig");
const ctl = @import("robot_control.zig");
const codecs = @import("codecs.zig");
const mjcf = @import("mjcf.zig");
const rmj = @import("robot_mjcf.zig");
const zp = @import("zimrphysics.zig");
const rphys = @import("robot_physics.zig");
const vec = zm.vec;
const go1_xml = @embedFile("tests/fixtures/robot/go1/go1.xml");

const sim_dt: f32 = 1.0 / 250.0;

pub fn main() !void {
    var da: std.heap.DebugAllocator(.{}) = .init;
    defer _ = da.deinit();
    const gpa: Allocator = da.allocator();
    const p = std.debug.print;

    var doc: codecs.xml.Document = try codecs.xml.parse(gpa, go1_xml, null);
    defer doc.deinit();
    var robot: mjcf.Robot = try mjcf.readRobot(gpa, &doc);
    defer robot.deinit();
    var imp: rmj.Imported = try rmj.build(gpa, &robot, .{
        .max_contacts = 64,
        .timestep = sim_dt,
        .gravity = vec(0, 0, -9.81),
    });
    defer imp.deinit();
    var m: rbt.Model = imp.model;
    var d: rbt.Data = try rbt.Data.init(gpa, &m);
    defer d.deinit();

    for (robot.keyframes) |key| {
        if (std.mem.eql(u8, key.name, "home")) _ = rmj.applyKeyframe(&m, &d, key);
    }
    d.stage = .stale;
    rbt.forward(&m, &d);

    var world: zp.World = try .init(gpa, 256);
    defer world.deinit(gpa);
    world.gravity = vec(0, 0, -9.81);
    const ground: zp.ShapeId = try world.shapes.add(gpa, .{
        .box = .{ .half_extent = vec(6, 6, 0.5), .convex_radius = 0.01 },
    });
    _ = try world.createBody(.{ .shape = ground, .position = vec(0, 0, -0.5), .motion_type = .static });
    var bridge: rphys.Bridge = try .init(gpa, &world, &m, &d, 256);
    defer bridge.deinit(&world);
    bridge.listen(&world);
    var act: ctl.Actuation = try ctl.Actuation.init(gpa, &m);
    defer act.deinit();

    var foot_body: [4]u32 = undefined;
    var foot_offset: [4]zm.Vec = undefined;
    for ([_][]const u8{ "FR_calf", "FL_calf", "RR_calf", "RL_calf" }, 0..) |name, i| {
        const b: u32 = imp.bodyIndex(name) orelse return error.NoFoot;
        foot_body[i] = b;
        var off: zm.Vec = vec(0, 0, 0);
        for (0..m.ngeom) |g| {
            if (m.geom_body[g] != b) continue;
            switch (m.geom_shape[g]) {
                .sphere => off = m.geom_pos[g],
                else => {},
            }
        }
        foot_offset[i] = off;
    }

    // Settle first — the question is about STEADY sliding, not the initial drop.
    const home: []f32 = try gpa.dupe(f32, d.pos[0..m.nq]);
    defer gpa.free(home);
    var warm: u32 = 0;
    while (warm < 750) : (warm += 1) {
        (ctl.PoseHold{ .target = home, .kp = 100, .kv = 6, .max_torque = 35 }).apply(&m, &d, act);
        rbt.forward(&m, &d);
        try bridge.sync(&world, &m, &d);
        try zp.step(&world, sim_dt);
        bridge.harvest(&d);
        rbt.step(&m, &d);
    }
    rbt.forward(&m, &d);

    var start: [4]zm.Vec = undefined;
    for (0..4) |i| {
        start[i] = d.body_xpos[foot_body[i]] + zm.rotate(d.body_xrot[foot_body[i]], foot_offset[i]);
    }
    const trunk_start: zm.Vec = d.body_xpos[1];

    p("perfectly static: home pose held, nothing commanded to move.\n\n", .{});

    var worst_cone: f32 = 0;
    var mean_cone: f64 = 0;
    var samples: u32 = 0;
    var peak_tang: f32 = 0;
    var tick: u32 = 0;
    while (tick < 5000) : (tick += 1) { // 20 s
        (ctl.PoseHold{ .target = home, .kp = 100, .kv = 6, .max_torque = 35 }).apply(&m, &d, act);
        rbt.forward(&m, &d);
        try bridge.sync(&world, &m, &d);
        try zp.step(&world, sim_dt);
        bridge.harvest(&d);
        rbt.step(&m, &d);
        for (0..d.contact_count) |c| {
            const base: usize = c * rbt.rows_per_contact;
            var normal: f32 = 0;
            for (0..rbt.rows_per_contact) |r| {
                normal += d.constraint_force[base + r];
            }
            if (normal < 1.0) {
                continue;
            }
            const t1: f32 = d.constraint_force[base] - d.constraint_force[base + 1];
            const t2: f32 = d.constraint_force[base + 2] - d.constraint_force[base + 3];
            const tang: f32 = @sqrt(t1 * t1 + t2 * t2);
            const mu: f32 = @max(0.01, d.contacts[c].friction[0]);
            const ratio: f32 = tang / (mu * normal);
            worst_cone = @max(worst_cone, ratio);
            peak_tang = @max(peak_tang, tang);
            mean_cone += ratio;
            samples += 1;
        }
    }
    rbt.forward(&m, &d);

    var slid: f32 = 0;
    for (0..4) |i| {
        const now: zm.Vec = d.body_xpos[foot_body[i]] +
            zm.rotate(d.body_xrot[foot_body[i]], foot_offset[i]);
        slid += zm.length3(now - start[i]);
    }
    const trunk_moved: f32 = zm.length3(d.body_xpos[1] - trunk_start);
    const mean: f32 = @floatCast(mean_cone / @as(f64, @floatFromInt(@max(1, samples))));

    p("  over 20 s of doing nothing:\n", .{});
    p("    total foot slide      {d:.4} m   ({d:.2} mm/s)\n", .{ slid, slid * 1000.0 / 20.0 });
    p("    trunk moved           {d:.4} m\n", .{trunk_moved});
    p("    friction demand       mean {d:.3}   worst {d:.3}   of the cone\n", .{ mean, worst_cone });
    p("    peak tangential force {d:.2} N\n\n", .{peak_tang});
    if (worst_cone < 0.8) {
        p("  ★ THE DEMAND IS WELL INSIDE THE CONE, so this is NOT Coulomb slip and MORE FRICTION\n", .{});
        p("    WOULD NOT FIX IT. The feet are creeping while the ground still has grip to spare.\n", .{});
    } else {
        p("  ★ THE DEMAND REACHES THE CONE, so friction is the binding limit here.\n", .{});
    }
}
