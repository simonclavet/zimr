//! Do re-planting and compliance fix the foot slip? One question, one file.
//!
//! ★ MEASURED FIRST: the shipped controller slides every foot 4.3 m in 12 s, with the friction
//! demand peaking at 1.83x the cone. Two cheap suspects, both to be tried before any planner:
//!
//!   * **RE-PLANT** — the foot targets are captured once at stance and never updated, so the
//!     moment the robot drifts every leg hauls toward a point that no longer exists;
//!   * **COMPLIANCE** — a softer leg demands less force for the same position error.
const std = @import("std");
const Allocator = std.mem.Allocator;
const zm = @import("zm");
const rbt = @import("robot.zig");
const ctl = @import("robot_control.zig");
const codecs = @import("codecs.zig");
const mjcf = @import("mjcf.zig");
const rmj = @import("robot_mjcf.zig");
const zp = @import("zimrphysics.zig");
const rphys = @import("robot_physics.zig");
const vec = zm.vec;
const go1_xml = @embedFile("tests/fixtures/robot/go1/go1.xml");

const sim_dt: f32 = 1.0 / 250.0;
const cone_rate: f32 = 0.55;
const roll_rate: f32 = 0.31;
const cone_amount: f32 = 0.22;
const roll_amount: f32 = 0.20;

fn commandAt(t: f32) [3]f32 {
    const cone: f32 = t * cone_rate * 2.0 * zm.pi;
    const rock: f32 = t * roll_rate * 2.0 * zm.pi;
    return .{ roll_amount * @sin(rock), cone_amount * @cos(cone), cone_amount * @sin(cone) };
}

pub fn main() !void {
    var da: std.heap.DebugAllocator(.{}) = .init;
    defer _ = da.deinit();
    const gpa: Allocator = da.allocator();
    const p = std.debug.print;

    var doc: codecs.xml.Document = try codecs.xml.parse(gpa, go1_xml, null);
    defer doc.deinit();
    var robot: mjcf.Robot = try mjcf.readRobot(gpa, &doc);
    defer robot.deinit();
    var imp: rmj.Imported = try rmj.build(gpa, &robot, .{
        .max_contacts = 64,
        .timestep = sim_dt,
        .gravity = vec(0, 0, -9.81),
    });
    defer imp.deinit();
    var m: rbt.Model = imp.model;
    var d: rbt.Data = try rbt.Data.init(gpa, &m);
    defer d.deinit();

    var applied: bool = false;
    for (robot.keyframes) |key| {
        if (std.mem.eql(u8, key.name, "home")) applied = rmj.applyKeyframe(&m, &d, key);
    }
    if (!applied) @memcpy(d.pos, m.qpos0);
    d.stage = .stale;
    rbt.forward(&m, &d);

    var world: zp.World = try .init(gpa, 256);
    defer world.deinit(gpa);
    world.gravity = vec(0, 0, -9.81);
    const ground: zp.ShapeId = try world.shapes.add(gpa, .{
        .box = .{ .half_extent = vec(6, 6, 0.5), .convex_radius = 0.01 },
    });
    _ = try world.createBody(.{ .shape = ground, .position = vec(0, 0, -0.5), .motion_type = .static });
    var bridge: rphys.Bridge = try .init(gpa, &world, &m, &d, 256);
    defer bridge.deinit(&world);
    bridge.listen(&world);

    var act: ctl.Actuation = try ctl.Actuation.init(gpa, &m);
    defer act.deinit();
    const ik_scratch: []zm.Vec = try gpa.alloc(zm.Vec, ctl.Ik.scratchSize(&m));
    defer gpa.free(ik_scratch);

    const home: []f32 = try gpa.dupe(f32, d.pos);
    defer gpa.free(home);
    var settle: u32 = 0;
    while (settle < 500) : (settle += 1) {
        (ctl.PoseHold{ .target = home, .kp = 400, .kv = 12, .max_torque = 35 }).apply(&m, &d, act);
        rbt.forward(&m, &d);
        try bridge.sync(&world, &m, &d);
        try zp.step(&world, sim_dt);
        bridge.harvest(&d);
        rbt.step(&m, &d);
    }
    rbt.forward(&m, &d);

    const stance: []f32 = try gpa.dupe(f32, d.pos);
    defer gpa.free(stance);
    const rest_rot: zm.Quat = d.body_xrot[1];
    const rest_height: f32 = d.pos[m.jnt_qpos_adr[0] + 2];

    var foot_body: [4]u32 = undefined;
    var foot_offset: [4]zm.Vec = undefined;
    var home_planted: [4]zm.Vec = undefined;
    var legs: [4]ctl.Actuation = undefined;
    const names = [_][]const u8{ "FR_calf", "FL_calf", "RR_calf", "RL_calf" };
    for (names, 0..) |name, i| {
        const body: u32 = imp.bodyIndex(name) orelse return error.NoFoot;
        foot_body[i] = body;
        legs[i] = try ctl.limbActuation(gpa, &m, body);
        var offset: zm.Vec = vec(0, 0, 0);
        for (0..m.ngeom) |g| {
            if (m.geom_body[g] != body) continue;
            switch (m.geom_shape[g]) {
                .sphere => offset = m.geom_pos[g],
                else => {},
            }
        }
        foot_offset[i] = offset;
        home_planted[i] = d.body_xpos[body] + zm.rotate(d.body_xrot[body], offset);
    }
    defer for (0..4) |i| legs[i].deinit();

    const target: []f32 = try gpa.alloc(f32, m.nq);
    defer gpa.free(target);
    const keep: []f32 = try gpa.alloc(f32, m.nq);
    defer gpa.free(keep);

    p("routine for 12 s. slip is total foot travel from where each foot started.\n\n", .{});
    p("     kp  replant   attitude lag   total slip   worst cone   ticks over\n", .{});

    // ── ★★★ THE DEMO'S ACTUAL SETTINGS ARE kp 100, kv 2 — NOT WHAT I HAD BEEN TESTING ──
    //
    // The quadruped has its own inline PD, and it ships at kp 100 with kv 2. My "baseline" of
    // kp 400 / kv 12 was copied from a different example and was never this demo's controller,
    // so a whole sweep measured something nobody runs.
    //
    // ★ AND kv 2 AGAINST kp 100 IS BADLY UNDERDAMPED — the ratio-preserving formula wants 6.
    // An underdamped leg oscillates about its target, and an oscillating foot is one that
    // alternately grips and breaks away. That is a much better explanation of "the feet slide"
    // than stiffness alone.
    for ([_][2]f32{
        .{ 100, 2 }, // exactly as the demo ships
        .{ 100, 6 }, // same stiffness, damping ratio corrected
        .{ 100, 12 }, // overdamped
        .{ 200, 8 },
        .{ 60, 5 },
    }) |cfg| {
        const stiffness: f32 = cfg[0];
        const damping: f32 = cfg[1];
        const replant: bool = false;

        @memcpy(d.pos, stance);
        @memset(d.vel, 0);
        @memset(d.applied_force, 0);
        d.stage = .stale;
        rbt.forward(&m, &d);
        bridge.teleported();
        @memcpy(target, stance);
        var planted: [4]zm.Vec = home_planted;

        var trim_roll: f32 = 0;
        var trim_pitch: f32 = 0;
        var lag_sum: f64 = 0;
        var lag_n: u32 = 0;
        var worst_cone: f32 = 0;
        var over: u32 = 0;
        var tick: u32 = 0;
        var clock: f32 = 0;
        while (clock < 12.0) : (clock += sim_dt) {
            const want = commandAt(clock);
            if (tick % 4 == 0) {
                const seen = zm.quatToEulerXYZ(zm.qmul(d.body_xrot[1], zm.conjugate(rest_rot)));
                trim_roll = zm.clamp(trim_roll + 0.5 * (want[0] - seen[0]), -0.4, 0.4);
                trim_pitch = zm.clamp(trim_pitch + 0.5 * (want[1] - seen[1]), -0.4, 0.4);

                // ★ RE-PLANT: the target becomes where the foot IS. A pinned target the robot has
                // already left is a force demand that only grows.
                if (replant) {
                    for (0..4) |i| {
                        planted[i] = d.body_xpos[foot_body[i]] +
                            zm.rotate(d.body_xrot[foot_body[i]], foot_offset[i]);
                    }
                }

                @memcpy(keep, d.pos[0..m.nq]);
                @memcpy(d.pos, stance);
                const root: u32 = m.jnt_qpos_adr[0];
                d.pos[root + 2] = rest_height;
                const dq: zm.Quat = zm.qmul(
                    zm.qmul(
                        zm.quatFromAxisAngle(vec(0, 0, 1), want[2]),
                        zm.quatFromAxisAngle(vec(0, 1, 0), want[1] + trim_pitch),
                    ),
                    zm.quatFromAxisAngle(vec(1, 0, 0), want[0] + trim_roll),
                );
                const wq: zm.Quat = zm.qmul(dq, rest_rot);
                d.pos[root + 3] = wq[0];
                d.pos[root + 4] = wq[1];
                d.pos[root + 5] = wq[2];
                d.pos[root + 6] = wq[3];
                d.stage = .stale;
                for (0..4) |i| {
                    _ = (ctl.Ik{ .max_iterations = 24 }).solve(
                        &m,
                        &d,
                        legs[i],
                        .{ .body = foot_body[i], .offset = foot_offset[i], .goal = planted[i] },
                        ik_scratch,
                    );
                }
                @memcpy(target, stance);
                for (0..m.njnt) |j| {
                    if (m.jnt_type[j] != .hinge) continue;
                    const q: u32 = m.jnt_qpos_adr[j];
                    target[q] = d.pos[q];
                }
                @memcpy(d.pos[0..m.nq], keep);
                d.stage = .stale;
                rbt.forward(&m, &d);
            }
            (ctl.PoseHold{
                .target = target,
                .kp = stiffness,
                // ★★★ THE DAMPING MUST MATCH THE SHIPPED CONTROLLER. The demo uses kp 400,
                // kv 12; the textbook `2√kp` gives 40 — three times as much, a different
                // controller, and it scored the "baseline" at 0.477 rad of lag where the real
                // one scores 0.0098. `0.6·√kp` reproduces 12 at kp 400 and holds the damping
                // RATIO fixed, so a stiffness sweep measures stiffness alone.
                .kv = damping,
                .max_torque = 35,
            }).apply(&m, &d, act);
            rbt.forward(&m, &d);
            try bridge.sync(&world, &m, &d);
            try zp.step(&world, sim_dt);
            bridge.harvest(&d);
            rbt.step(&m, &d);
            tick += 1;

            if (clock > 2.0) {
                const seen = zm.quatToEulerXYZ(zm.qmul(d.body_xrot[1], zm.conjugate(rest_rot)));
                lag_sum += @sqrt(
                    (want[0] - seen[0]) * (want[0] - seen[0]) +
                        (want[1] - seen[1]) * (want[1] - seen[1]),
                );
                lag_n += 1;
                for (0..d.contact_count) |c| {
                    const base: usize = c * rbt.rows_per_contact;
                    var normal: f32 = 0;
                    for (0..rbt.rows_per_contact) |r| normal += d.constraint_force[base + r];
                    if (normal < 1.0) continue;
                    const t1: f32 = d.constraint_force[base + 0] - d.constraint_force[base + 1];
                    const t2: f32 = d.constraint_force[base + 2] - d.constraint_force[base + 3];
                    const tang: f32 = @sqrt(t1 * t1 + t2 * t2);
                    const ratio: f32 = tang / (@max(0.01, d.contacts[c].friction[0]) * normal);
                    worst_cone = @max(worst_cone, ratio);
                    if (ratio > 0.98) over += 1;
                }
            }
        }
        rbt.forward(&m, &d);
        var slip: f32 = 0;
        for (0..4) |i| {
            const now: zm.Vec = d.body_xpos[foot_body[i]] +
                zm.rotate(d.body_xrot[foot_body[i]], foot_offset[i]);
            slip += zm.length3(now - home_planted[i]);
        }
        const lag: f32 = @floatCast(lag_sum / @as(f64, @floatFromInt(@max(1, lag_n))));
        p("  {d:>5.0}  {any:<7}  {d:>12.4}  {d:>11.3}  {d:>11.2}  {d:>11}\n", .{
            stiffness, replant, lag, slip, worst_cone, over,
        });
    }
}
