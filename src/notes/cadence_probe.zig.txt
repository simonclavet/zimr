//! What does the replan cadence actually cost? One question, one file.
//!
//! ★ THE DEMO REPLANS EVERY FOURTH SIM STEP with `feedbackControl` covering the gaps. That is
//! what real MPC does and what `feedbackControl` exists for — but "supposed to" is not a
//! measurement, and it was shipped labelled as unverified. This settles it.
const std = @import("std");
const Allocator = std.mem.Allocator;
const zm = @import("zm");
const rbt = @import("robot.zig");
const ctl = @import("robot_control.zig");
const mpc = @import("robot_mpc.zig");
const codecs = @import("codecs.zig");
const mjcf = @import("mjcf.zig");
const rmj = @import("robot_mjcf.zig");
const vec = zm.vec;
const float = zm.float;
const tracker_xml = @embedFile("tests/fixtures/robot/tracker.xml");

const sim_dt: f32 = 1.0 / 250.0;
const knots: u32 = 75;
const speed: f32 = 1.5;

fn targetAt(t: f32) zm.Vec {
    const a: f32 = t * speed;
    return vec(0.95 + 0.32 * @cos(a), 0.34 * @sin(a), 1.15 + 0.20 * @sin(a * 0.5));
}

pub fn main() !void {
    var da: std.heap.DebugAllocator(.{}) = .init;
    defer _ = da.deinit();
    const gpa: Allocator = da.allocator();

    var doc: codecs.xml.Document = try codecs.xml.parse(gpa, tracker_xml, null);
    defer doc.deinit();
    var robot: mjcf.Robot = try mjcf.readRobot(gpa, &doc);
    defer robot.deinit();
    var imp: rmj.Imported = try rmj.build(gpa, &robot, .{
        .max_contacts = 8,
        .timestep = sim_dt,
        .gravity = vec(0, 0, -9.81),
    });
    defer imp.deinit();
    var m: rbt.Model = imp.model;
    var d: rbt.Data = try rbt.Data.init(gpa, &m);
    defer d.deinit();
    rbt.forward(&m, &d);
    const grip: u32 = imp.bodyIndex("gripper") orelse return error.NoGripper;
    const rest: []f32 = try gpa.dupe(f32, d.pos);
    defer gpa.free(rest);

    var arm_act: ctl.Actuation = try ctl.limbActuation(gpa, &m, grip);
    defer arm_act.deinit();
    const ik_scratch: []zm.Vec = try gpa.alloc(zm.Vec, ctl.Ik.scratchSize(&m));
    defer gpa.free(ik_scratch);

    // The same pose table the demo uses, built the same chained way.
    const table_size: u32 = 512;
    const period: f32 = 2.0 * zm.pi / speed;
    const table: []f32 = try gpa.alloc(f32, table_size * m.nq);
    defer gpa.free(table);
    {
        @memcpy(d.pos, rest);
        d.stage = .stale;
        rbt.forward(&m, &d);
        var pass: u32 = 0;
        while (pass < 2) : (pass += 1) {
            for (0..table_size) |i| {
                const t: f32 = period * float(i) / float(table_size);
                _ = (ctl.Ik{ .max_iterations = 24, .max_step = 0.35 }).solve(&m, &d, arm_act, .{ .body = grip, .offset = vec(0, 0, 0), .goal = targetAt(t) }, ik_scratch);
                if (pass == 1) @memcpy(table[i * m.nq ..][0..m.nq], d.pos[0..m.nq]);
            }
        }
    }
    const aim: []f32 = try gpa.alloc(f32, m.nq);
    defer gpa.free(aim);

    var plan: mpc.Plan = try mpc.Plan.init(gpa, &m, knots);
    defer plan.deinit();
    const nstate: u32 = m.nq + m.nv;
    const reference: []f32 = try gpa.alloc(f32, (knots + 1) * nstate);
    defer gpa.free(reference);
    const state_w: []f32 = try gpa.alloc(f32, plan.ndx);
    defer gpa.free(state_w);
    const term_w: []f32 = try gpa.alloc(f32, plan.ndx);
    defer gpa.free(term_w);
    const ctrl_w: []f32 = try gpa.alloc(f32, m.nu);
    defer gpa.free(ctrl_w);
    for (0..m.nv) |i| {
        state_w[i] = 8.0;
        state_w[m.nv + i] = 0.05;
        term_w[i] = 40.0;
        term_w[m.nv + i] = 1.0;
    }
    @memset(ctrl_w, 1.0e-5);

    std.log.warn("replan cadence, 75 knots, 6 iterations, pose table (no runtime IK)", .{});
    std.log.warn("  cadence   mean err (m)   worst    replans/sec", .{});
    for ([_]u32{ 1, 2, 4, 8 }) |cadence| {
        @memcpy(d.pos, rest);
        @memset(d.vel, 0);
        @memset(d.ctrl, 0);
        @memset(d.applied_force, 0);
        d.stage = .stale;
        rbt.forward(&m, &d);
        @memset(plan.ctrl, 0);

        var total: f64 = 0;
        var worst: f32 = 0;
        var samples: u32 = 0;
        var tick: u32 = 0;
        var elapsed: f32 = 0;
        while (elapsed < 6.0) : (elapsed += sim_dt) {
            if (tick % cadence == 0) {
                for (0..plan.horizon + 1) |k| {
                    const when: f32 = elapsed + float(k) * sim_dt;
                    const phase: f32 = @mod(when / period, 1.0) * float(table_size);
                    const lo: u32 = @min(table_size - 1, zm.floori(u32, phase));
                    const hi: u32 = (lo + 1) % table_size;
                    const frac: f32 = phase - float(lo);
                    for (0..m.nq) |q| {
                        aim[q] = table[lo * m.nq + q] +
                            frac * (table[hi * m.nq + q] - table[lo * m.nq + q]);
                    }
                    @memcpy(reference[k * nstate ..][0..m.nq], aim);
                    @memset(reference[k * nstate + m.nq ..][0..m.nv], 0);
                }
                _ = mpc.optimize(&m, &d, &plan, .{
                    .state = state_w,
                    .control = ctrl_w,
                    .terminal = term_w,
                    .reference = reference,
                }, .{ .iterations = 6 });
            }
            mpc.feedbackControl(&m, &d, &plan, d.ctrl[0..m.nu]);
            mpc.shift(&plan);
            tick += 1;
            rbt.step(&m, &d);
            rbt.forward(&m, &d);
            if (elapsed > 1.5) {
                const err: f32 = zm.length3(d.body_xpos[grip] - targetAt(elapsed));
                total += err;
                worst = @max(worst, err);
                samples += 1;
            }
        }
        const mean: f32 = @floatCast(total / @as(f64, @floatFromInt(@max(1, samples))));
        std.log.warn("  {d:>7}   {d:>12.5}   {d:>6.4}   {d:>10.0}", .{ cadence, mean, worst, 250.0 / float(cadence) });
    }
}
