//! Does the catch close with more iterations? One question, one file.
//!
//! ★ `Result` already said the solver is STARVED rather than stuck: it spends its whole budget
//! every replan, reports `converged false`, and sits at the regularisation floor — so it is
//! still improving when it stops. This asks the only question that follows: how much more does
//! it need, and can a frame afford it?
const std = @import("std");
const Allocator = std.mem.Allocator;
const zm = @import("zm");
const rbt = @import("robot.zig");
const ctl = @import("robot_control.zig");
const mpc = @import("robot_mpc.zig");
const codecs = @import("codecs.zig");
const mjcf = @import("mjcf.zig");
const rmj = @import("robot_mjcf.zig");
const vec = zm.vec;
const keeper_xml = @embedFile("tests/fixtures/robot/keeper.xml");

const gravity: f32 = 9.81;
const sim_dt: f32 = 1.0 / 250.0;
const horizon: u32 = 220;

fn ballAt(p0: zm.Vec, v0: zm.Vec, t: f32) zm.Vec {
    return p0 + v0 * zm.splat(t) + vec(0, 0, -0.5 * gravity * t * t);
}
fn ballVel(v0: zm.Vec, t: f32) zm.Vec {
    return v0 + vec(0, 0, -gravity * t);
}

/// One throw, one iteration budget, one answer.
fn attempt(
    gpa: Allocator,
    m: *rbt.Model,
    d: *rbt.Data,
    hand: u32,
    rest: []const f32,
    arm_act: ctl.Actuation,
    ik_scratch: []zm.Vec,
    p0: zm.Vec,
    v0: zm.Vec,
    budget: u32,
    nearest_branch: bool,
    vel_weight: f32,
) !struct { gap: f32, relative: f32, replans: u32 } {
    // Interception: the latest instant Ik agrees is reachable.
    var meet_t: f32 = -1;
    var t: f32 = 0.15;
    while (t < 1.4) : (t += 0.02) {
        const at: zm.Vec = ballAt(p0, v0, t);
        if (at[2] < 0.3) break;
        @memcpy(d.pos, rest);
        d.stage = .stale;
        rbt.forward(m, d);
        const r: ctl.Ik.Result = (ctl.Ik{ .max_iterations = 60 }).solve(m, d, arm_act, .{ .body = hand, .offset = vec(0, 0, 0), .goal = at }, ik_scratch);
        if (r.reached) meet_t = t;
    }
    if (meet_t <= 0) return .{ .gap = 1.0e9, .relative = 0, .replans = 0 };

    @memcpy(d.pos, rest);
    d.stage = .stale;
    rbt.forward(m, d);
    _ = (ctl.Ik{ .max_iterations = 80 }).solve(m, d, arm_act, .{ .body = hand, .offset = vec(0, 0, 0), .goal = ballAt(p0, v0, meet_t) }, ik_scratch);
    const meet_pose: []f32 = try gpa.dupe(f32, d.pos[0..m.nq]);
    defer gpa.free(meet_pose);

    // Joint velocities putting the hand on the ball's velocity there.
    const meet_qvel: []f32 = try gpa.alloc(f32, m.nv);
    defer gpa.free(meet_qvel);
    @memset(meet_qvel, 0);
    {
        const jac: []zm.Vec = try gpa.alloc(zm.Vec, m.nv);
        defer gpa.free(jac);
        rbt.jacPoint(m, d, hand, d.body_xpos[hand], jac, null);
        var gram: [9]f32 = @splat(0);
        inline for (0..3) |a| {
            inline for (0..3) |b| {
                var sum: f32 = 0;
                for (0..m.nv) |v| sum += jac[v][a] * jac[v][b];
                gram[a * 3 + b] = sum;
            }
        }
        const ridge: f32 = 1.0e-4 * (@abs(gram[0]) + @abs(gram[4]) + @abs(gram[8]) + 1.0);
        gram[0] += ridge;
        gram[4] += ridge;
        gram[8] += ridge;
        const want: zm.Vec = ballVel(v0, meet_t);
        var lambda: [3]f32 = .{ want[0], want[1], want[2] };
        if (ctl.solve3Pub(&gram, &lambda)) {
            for (0..m.nv) |v| {
                meet_qvel[v] = lambda[0] * jac[v][0] + lambda[1] * jac[v][1] + lambda[2] * jac[v][2];
            }
        }
    }

    var plan: mpc.Plan = try mpc.Plan.init(gpa, m, horizon); // limits are on by default now
    defer plan.deinit();
    const nstate: u32 = m.nq + m.nv;
    const reference: []f32 = try gpa.alloc(f32, (horizon + 1) * nstate);
    defer gpa.free(reference);
    const state_w: []f32 = try gpa.alloc(f32, plan.ndx);
    defer gpa.free(state_w);
    const term_w: []f32 = try gpa.alloc(f32, plan.ndx);
    defer gpa.free(term_w);
    const ctrl_w: []f32 = try gpa.alloc(f32, m.nu);
    defer gpa.free(ctrl_w);
    for (0..m.nv) |i| {
        state_w[i] = 0.5;
        state_w[m.nv + i] = vel_weight;
        term_w[i] = 60.0;
        term_w[m.nv + i] = 4.0;
    }
    @memset(ctrl_w, 1.0e-4);

    @memcpy(d.pos, rest);
    @memset(d.vel, 0);
    @memset(d.ctrl, 0);
    d.stage = .stale;
    rbt.forward(m, d);
    @memset(plan.ctrl, 0);

    const scratch_pose: []f32 = try gpa.alloc(f32, m.nq);
    defer gpa.free(scratch_pose);
    const live_pose: []f32 = try gpa.alloc(f32, m.nq);
    defer gpa.free(live_pose);

    var gap: f32 = 1.0e9;
    var relative: f32 = 0;
    var replans: u32 = 0;
    var elapsed: f32 = 0;
    var tick: u32 = 0;
    while (elapsed < meet_t + 0.15) : (tick += 1) {
        if (tick % 5 == 0) {
            const left: f32 = @max(0.0, meet_t - elapsed);
            const arrive: u32 = @min(horizon, @as(u32, @intFromFloat(left / sim_dt)));
            // ── ★★★ THE TERMINAL KNOT *IS* THE INTERCEPTION ──
            //
            // With a fixed 0.88 s horizon the terminal knot — carrying by far the largest weight,
            // 60 against 0.5 — sat after the ball had already hit the floor. The plan was being
            // asked with maximum emphasis to be somewhere at a moment that no longer meant
            // anything, which is why iterating harder changed nothing at all.
            //
            // Shrinking the horizon to the time remaining puts that weight exactly where the
            // catch is, and keeps it there as the moment approaches.
            plan.setHorizon(@max(2, arrive));
            for (0..plan.horizon + 1) |k| {
                const along: f32 = if (arrive == 0) 1.0 else @min(
                    1.0,
                    @as(f32, @floatFromInt(k)) / @as(f32, @floatFromInt(arrive)),
                );
                const blend: f32 = along * along * (3.0 - 2.0 * along);
                for (0..m.nq) |q| {
                    reference[k * nstate + q] = d.pos[q] + blend * (meet_pose[q] - d.pos[q]);
                }
                const near_end: bool = arrive > 0 and k + 10 >= arrive;
                for (0..m.nv) |v| {
                    reference[k * nstate + m.nq + v] = if (near_end) meet_qvel[v] else 0;
                }
            }
            // ── ★★★ IK FROM WHERE THE ARM IS, NOT FROM ITS REST POSE ──
            //
            // The interception pose was solved ONCE from `rest`, so it is an arbitrary branch of
            // a redundant arm — one of many configurations that put the hand on the ball, chosen
            // before the arm had moved. The planner was then forced onto that exact branch, with
            // the whole null space thrown away.
            //
            // Re-solving from the CURRENT pose returns the nearest configuration that does the
            // job. `Ik` takes least-norm steps, so "nearest" is what it naturally finds — the
            // redundancy gets used instead of discarded, without the cost machinery needing to
            // understand the task at all.
            if (nearest_branch) {
                @memcpy(scratch_pose, d.pos[0..m.nq]);
                _ = (ctl.Ik{ .max_iterations = 40 }).solve(
                    m,
                    d,
                    arm_act,
                    .{ .body = hand, .offset = vec(0, 0, 0), .goal = ballAt(p0, v0, meet_t) },
                    ik_scratch,
                );
                @memcpy(live_pose, d.pos[0..m.nq]);
                @memcpy(d.pos[0..m.nq], scratch_pose);
                d.stage = .stale;
                rbt.forward(m, d);
                for (0..plan.horizon + 1) |k| {
                    const along: f32 = if (arrive == 0) 1.0 else @min(
                        1.0,
                        @as(f32, @floatFromInt(k)) / @as(f32, @floatFromInt(arrive)),
                    );
                    const blend: f32 = along * along * (3.0 - 2.0 * along);
                    for (0..m.nq) |q| {
                        reference[k * nstate + q] = d.pos[q] + blend * (live_pose[q] - d.pos[q]);
                    }
                }
            }
            _ = mpc.optimize(m, d, &plan, .{
                .state = state_w,
                .control = ctrl_w,
                .terminal = term_w,
                .reference = reference,
            }, .{ .iterations = budget });
            replans += 1;
        }
        @memcpy(d.ctrl, plan.ctrl[0..m.nu]);
        mpc.shift(&plan);
        rbt.step(m, d);
        elapsed += sim_dt;

        const ball: zm.Vec = ballAt(p0, v0, elapsed);
        const now: f32 = zm.length3(d.body_xpos[hand] - ball);
        if (now < gap) {
            gap = now;
            relative = zm.length3(d.cvel[hand].lin - ballVel(v0, elapsed));
        }
    }
    return .{ .gap = gap, .relative = relative, .replans = replans };
}

pub fn main() !void {
    var da: std.heap.DebugAllocator(.{}) = .init;
    defer _ = da.deinit();
    const gpa: Allocator = da.allocator();
    const p = std.debug.print;

    var doc: codecs.xml.Document = try codecs.xml.parse(gpa, keeper_xml, null);
    defer doc.deinit();
    var robot: mjcf.Robot = try mjcf.readRobot(gpa, &doc);
    defer robot.deinit();
    var imp: rmj.Imported = try rmj.build(gpa, &robot, .{
        .max_contacts = 16,
        .timestep = sim_dt,
        .gravity = vec(0, 0, -gravity),
    });
    defer imp.deinit();
    var m: rbt.Model = imp.model;
    var d: rbt.Data = try rbt.Data.init(gpa, &m);
    defer d.deinit();
    rbt.forward(&m, &d);
    const hand: u32 = imp.bodyIndex("hand") orelse return error.NoHand;
    const rest: []f32 = try gpa.dupe(f32, d.pos);
    defer gpa.free(rest);
    var arm_act: ctl.Actuation = try ctl.limbActuation(gpa, &m, hand);
    defer arm_act.deinit();
    const ik_scratch: []zm.Vec = try gpa.alloc(zm.Vec, ctl.Ik.scratchSize(&m));
    defer gpa.free(ik_scratch);

    const throws = [_]struct { p0: zm.Vec, v0: zm.Vec }{
        .{ .p0 = vec(-2.6, 0.0, 1.0), .v0 = vec(3.4, 0.0, 3.0) },
        .{ .p0 = vec(-2.4, -0.9, 0.9), .v0 = vec(3.2, 1.1, 3.3) },
        .{ .p0 = vec(-2.8, 0.7, 1.2), .v0 = vec(3.6, -0.9, 2.7) },
        .{ .p0 = vec(-2.2, 0.4, 0.8), .v0 = vec(3.0, -0.5, 3.6) },
    };

    p("does the catch close with more iterations?\n\n", .{});

    p("  nearest branch | vel weight |  gaps                              catches\n", .{});
    for ([_]bool{ false, true }) |nearest| {
        for ([_]f32{ 0.02, 0.5 }) |vw| {
            const budget: u32 = 8;
            var caught: u32 = 0;
            for (throws, 0..) |thrown, index| {
                const r = try attempt(gpa, &m, &d, hand, rest, arm_act, ik_scratch, thrown.p0, thrown.v0, budget, nearest, vw);
                const ok: bool = r.gap < 0.05 and r.relative < 0.5;
                if (ok) caught += 1;
                p("{d:>8.3}", .{r.gap});
                _ = index;
            }
            p("   -> {d} of {d}   ({s}, vel weight {d:.2})\n", .{ caught, throws.len, if (nearest) "nearest branch" else "fixed branch  ", vw });
        }
    }
}
